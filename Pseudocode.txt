    int n = arr.length
    list(list(float)) buckets = new list of n empty lists

// Merge Sort Algorithm
FUNCTION mergeSort(array(comparable) arr)

    // Base case: an array of 0 or 1 elements is already sorted
    IF arr.length <= 1
        RETURN arr
    END IF

    // Find the middle index to divide the array into two halves
    int mid = arr.length / 2

    // Recursively sort the two halves
    array(comparable) leftHalf = mergeSort(arr[0..mid-1])
    array(comparable) rightHalf = mergeSort(arr[mid..arr.length-1])

    // Merge the now-sorted halves
    array(comparable) result = merge(leftHalf, rightHalf)

    RETURN result
END FUNCTION



// Merge Sort Algorithm
//(initial input would be: mergeSort(arr, 0, arr.length - 1))
FUNCTION mergeSort(array(float) arr, int left, int right)
    
    IF left < right 
        
        // Find the mid point in the array
        int mid = left + (right - left) / 2
        
        // Sort first and second half
        mergeSort(arr, left, mid)
        mergeSort(arr, mid + 1, right)
        
        // Merge sorted halves
        merge(arr, left, mid, right)
        
    END IF
END FUNCTION


// Helper function to merge two sorted subarrays
FUNCTION merge(array(float) arr, int left, int mid, int right)
    
    // Find lengths of the two sub arrays which will be created
    int n1 = mid - left + 1
    int n2 = right - mid
    
    // Create temporary arrays
    array(float) L = new array of size n1
    array(float) R = new array of size n2
    
    // Copy data to temporary arrays from input arr
    FOR i FROM 0 TO n1 - 1
        L[i] = arr[left + i]
    END FOR
    
    FOR j FROM 0 TO n2 - 1
        R[j] = arr[mid + 1 + j]
    END FOR
        
    // Initial indexes of first and second subarrays
    int i = 0
    int j = 0
    
    // Initial index of merged subarray
    int k = left
    
    WHILE i < n1 AND j < n2 
        IF L[i] <= R[j]
            arr[k] = L[i]
            i = i + 1
        ELSE
            arr[k] = R[j]
            j = j + 1
        END IF
        k = k + 1
    END WHILE
    
    // Copy remaining elements of L[] (if any)
    WHILE i < n1 
        arr[k] = L[i]
        i = i + 1
        k = k + 1
    END WHILE
    
    // Copy remaining elements of R[] (if any)
    WHILE j < n2 
        arr[k] = R[j]
        j = j + 1
        k = k + 1
    END WHILE
    
END FUNCTION



// Linear Search Algorithm
FUNCTION linearSearch(array(float) arr, float target)

    // Iterate through every element in the array
    FOR i FROM 0 TO arr.length - 1 
        // Check if the current element matches the target
        IF arr[i] == target 
            RETURN i  // Target found, return its index
        END IF
    END FOR
END FUNCTION




// Binary Search Algorithm (Assumes the input array is already sorted)
FUNCTION binarySearch(array(float) arr, float target)

    // Initialize the search boundaries to the entire array
    int low = 0
    int high = arr.length - 1

    // Continue searching while the search space is valid
    WHILE low <= high 

        // Find the middle element of the current search space
        int mid = (low + high) / 2
        float midValue = arr[mid]

        IF target == midValue THEN
            RETURN mid        // Target found at index 'mid'
        ELSE IF target < midValue THEN
            high = mid - 1    // Discard the right half
        ELSE
            low = mid + 1     // Discard the left half
        END IF

    END WHILE

    RETURN -1  // Target was not found in the array
END FUNCTION




// Dijkstra's Algorithm for finding shortest paths from a start vertex
FUNCTION dijkstra(Graph graph, string startVertexData)

    // Find the index of the starting vertex in the vertex data array
    int start_vertex = graph.vertex_data.indexOf(startVertexData)
    
    //Initialize distances array with infinity for all vertices
    array(float) distances = new array of size graph.size
    FOR i FROM 0 TO graph.size - 1
        distances[i] = INFINITY
    END FOR

    distances[start_vertex] = 0 // Set first to 0 (very important)
    
    // Initialize visited array to track processed vertices
    array(boolean) visited = new array of size graph.size
    FOR i FROM 0 TO graph.size - 1
        visited[i] = false
    END FOR

    // Process all vertices
    FOR count FROM 0 TO graph.size - 1
        
        // Find the unvisited vertex with the smallest distance
        float min_distance = INFINITY
        int u = -1  // will store the index of the selected vertex
        
        FOR i FROM 0 TO graph.size - 1
            IF visited[i] == false AND distances[i] < min_distance THEN
                min_distance = distances[i]
                u = i
            END IF
        END FOR
        
        // If no suitable vertex found, exit early
        IF u == -1 THEN
            BREAK
        END IF
        
        // Mark the selected vertex as visited
        visited[u] = true
        
        // Update distances to all adjacent vertices
        FOR v FROM 0 TO graph.size - 1
            // Check if there's an edge and vertex is unvisited
            IF graph.adj_matrix[u][v] != 0 AND visited[v] == false THEN
                float alt = distances[u] + graph.adj_matrix[u][v]
                IF alt < distances[v] THEN
                    distances[v] = alt
                END IF
            END IF
        END FOR
        
    END FOR
    
    RETURN distances
END FUNCTION
