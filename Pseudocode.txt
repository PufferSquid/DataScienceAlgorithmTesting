// Merge Sort Algorithm
//(initial input would be: mergeSort(arr, 0, arr.length - 1))
FUNCTION mergeSort(array(float) arr, int left, int right)
    
    IF left < right 
        
        // Find the mid point in the array
        int mid = left + (right - left) / 2
        
        // Sort first and second half
        mergeSort(arr, left, mid)
        mergeSort(arr, mid + 1, right)
        
        // Merge sorted halves
        merge(arr, left, mid, right)
        
    END IF
END FUNCTION


// Helper function to merge two sorted subarrays
FUNCTION merge(array(float) arr, int left, int mid, int right)
    
    // Find lengths of the two sub arrays which will be created
    int n1 = mid - left + 1
    int n2 = right - mid
    
    // Create temporary arrays
    array(float) L = new array of size n1
    array(float) R = new array of size n2
    
    // Copy data to temporary arrays from input arr
    FOR i FROM 0 TO n1 - 1
        L[i] = arr[left + i]
    END FOR
    
    FOR i FROM 0 TO n2 - 1
        R[i] = arr[mid + 1 + i]
    END FOR
        
    // Initial indexes of first and second subarrays
    int i = 0
    int j = 0
    
    // Initial index of merged subarray
    int k = left
    
    WHILE i < n1 AND j < n2 
        IF L[i] <= R[j]
            arr[k] = L[i]
            i = i + 1
        ELSE
            arr[k] = R[j]
            j = j + 1
        END IF
        k = k + 1
    END WHILE
    
    // Copy remaining elements of L[] (if any)
    WHILE i < n1 
        arr[k] = L[i]
        i = i + 1
        k = k + 1
    END WHILE
    
    // Copy remaining elements of R[] (if any)
    WHILE j < n2 
        arr[k] = R[j]
        j = j + 1
        k = k + 1
    END WHILE
    
END FUNCTION



// Linear Search Algorithm
FUNCTION linearSearch(array(float) arr, float target)

    // Iterate through every element in the array
    FOR i FROM 0 TO arr.length - 1 
        // Check if the current element matches the target
        IF arr[i] == target 
            RETURN i  // Target found, return its index
        END IF
    END FOR
END FUNCTION




// Binary Search Algorithm (Assumes the input array is already sorted)
FUNCTION binarySearch(array(float) arr, float target)

    // Initialize the search boundaries to the entire array
    int low = 0
    int high = arr.length - 1

    // Continue searching while the search space is valid
    WHILE low <= high 

        // Find the middle element of the current search space
        int mid = (low + high) / 2
        float midValue = arr[mid]

        IF target == midValue THEN
            RETURN mid // Target found at index 'mid'
        ELSE IF target < midValue THEN
            high = mid - 1 // Discard the right half
        ELSE
            low = mid + 1// Discard the left half
        END IF

    END WHILE

    RETURN -1  // Target was not found in the array
END FUNCTION




// Dijkstra's Algorithm for finding shortest paths from a start vertex
FUNCTION Dijkstra(dict graph, str startNode)

    // Initialize all distances to infinity and the start node distance to 0
    dict distances = new empty dictionary
    FOR EACH node IN graph.keys
        distances[node] = INFINITY
    END FOR
    distances[startNode] = 0
    
    // Initialize an empty set to track visited nodes
    set visited = EMPTY_SET
    
    // Iterate through all nodes in the graph
    FOR i = 0 TO graph.length - 1
        
        // Find the unvisited node with the smallest distance
        float minDistance = INFINITY
        str currentNode = NULL
        
        FOR EACH node IN graph.keys
            IF node NOT IN visited AND distances[node] < minDistance THEN
                minDistance = distances[node]
                currentNode = node
            END IF
        END FOR
        
        // If no valid node is found, break the loop
        IF currentNode == NULL THEN
            BREAK
        END IF
        
        // Mark the current node as visited
        visited.add(currentNode)
        
        // Update distances for neighboring nodes
        FOR EACH neighbor, cost IN graph[currentNode].items
            IF neighbor NOT IN visited THEN
                float newDistance = distances[currentNode] + cost
                
                IF newDistance < distances[neighbor] THEN
                    distances[neighbor] = newDistance
                END IF
            END IF
        END FOR
        
    END FOR
    
    RETURN distances  // Return shortest distances from start node
END FUNCTION
